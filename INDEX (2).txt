    <script>
        // અહીં તમારી Google Apps Script ની Web App URL મૂકો
        const scriptURL = 'YOUR_WEB_APP_URL';
        const form = document.forms['dataForm'];
        const statusText = document.getElementById('status');

        form.addEventListener('submit', e => {
            e.preventDefault();
            statusText.style.color = 'blue';
            statusText.innerHTML = 'સબમિટ થઈ રહ્યું છે...';

            fetch(scriptURL, { method: 'POST', body: new FormData(form)})
                .then(response => {
                    statusText.style.color = 'green';
                    statusText.innerHTML = 'ડેટા સફળતાપૂર્વક Google Sheet માં સેવ થઈ ગયો છે!';
                    form.reset();
                })
                .catch(error => {
                    statusText.style.color = 'red';
                    statusText.innerHTML = 'ભૂલ આવી! ફરી પ્રયાસ કરો.';
                    console.error('Error!', error.message);
                });
        });
    </script>
